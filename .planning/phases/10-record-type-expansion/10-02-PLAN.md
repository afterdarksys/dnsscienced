Consider a helper function to reduce duplication:
```go
func assertRoundTrip(t *testing.T, z *Zone, z2 *Zone, owner string, rrtype uint16) {
    t.Helper()
    orig := z.GetRecords(owner, rrtype)
    loaded := z2.GetRecords(owner, rrtype)
    if len(orig) != len(loaded) {
        t.Fatalf("record count mismatch for %s type %d: %d vs %d", owner, rrtype, len(orig), len(loaded))
    }
    for i := range orig {
        buf1, buf2 := make([]byte, dns.MaxMsgSize), make([]byte, dns.MaxMsgSize)
        off1, _ := dns.PackRR(orig[i], buf1, 0, nil, false)
        off2, _ := dns.PackRR(loaded[i], buf2, 0, nil, false)
        if !bytes.Equal(buf1[:off1], buf2[:off2]) {
            t.Errorf("wire mismatch for record %d of %s type %d", i, owner, rrtype)
        }
    }
}
```
