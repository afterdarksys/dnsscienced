```go
// atomicKeySet holds an API key set that can be hot-swapped without a mutex.
// The interceptor calls Load() on every request (lock-free pointer load).
// The SIGHUP handler or ReloadAPIKeys calls Store() to swap in a new set.
type atomicKeySet struct {
    v atomic.Value // stores map[string]struct{}
}

func newAtomicKeySet(keys []string) *atomicKeySet {
    s := &atomicKeySet{}
    s.Store(keys)
    return s
}

// Store atomically replaces the key set. Safe to call from any goroutine.
func (a *atomicKeySet) Store(keys []string) {
    m := make(map[string]struct{}, len(keys))
    for _, k := range keys {
        m[k] = struct{}{}
    }
    a.v.Store(m)
}

// Load returns the current key set. Panics if Store has never been called.
func (a *atomicKeySet) Load() map[string]struct{} {
    return a.v.Load().(map[string]struct{})
}
```

The existing `apiKeyUnaryInterceptor` and `apiKeyStreamInterceptor` still use the old local `map[string]struct{}` — they will be replaced in Plan 02. Do not change them yet.
