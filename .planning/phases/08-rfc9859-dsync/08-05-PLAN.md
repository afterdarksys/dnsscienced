    ```go
    import "net"

    // SourceACL validates inbound NOTIFY source IPs against a CIDR allowlist.
    // A nil or empty allowlist means all sources are accepted (per D-05).
    type SourceACL struct {
        networks []*net.IPNet
        allowAll bool
    }

    // NewSourceACL creates an ACL from a list of CIDR strings (e.g., "10.0.0.0/8")
    // or single IPs (e.g., "203.0.113.5" — treated as /32 or /128).
    // Empty or nil slice means allow all.
    func NewSourceACL(cidrs []string) *SourceACL {
        if len(cidrs) == 0 {
            return &SourceACL{allowAll: true}
        }
        var nets []*net.IPNet
        for _, cidr := range cidrs {
            _, network, err := net.ParseCIDR(cidr)
            if err != nil {
                // Try as single IP
                ip := net.ParseIP(cidr)
                if ip == nil {
                    continue // skip invalid entries
                }
                if ip.To4() != nil {
                    _, network, _ = net.ParseCIDR(cidr + "/32")
                } else {
                    _, network, _ = net.ParseCIDR(cidr + "/128")
                }
            }
            if network != nil {
                nets = append(nets, network)
            }
        }
        return &SourceACL{networks: nets, allowAll: len(nets) == 0}
    }

    // Check returns true if the IP is allowed by this ACL.
    func (a *SourceACL) Check(ip net.IP) bool {
        if a.allowAll {
            return true
        }
        for _, n := range a.networks {
            if n.Contains(ip) {
                return true
            }
        }
        return false
    }
    ```

    ```go
    import (
        "bytes"
        "encoding/base64"
        "encoding/json"
        "fmt"
        "net/http"
        "time"
    )

    // WebhookPayload is the JSON body sent on NOTIFY events (per D-03).
    type WebhookPayload struct {
        Zone      string    `json:"zone"`
        Qtype     string    `json:"qtype"`
        SourceIP  string    `json:"source_ip"`
        Timestamp time.Time `json:"timestamp"`
    }

    // WebhookClient fires POST requests on inbound NOTIFY events.
    // Fire-and-forget: 5s timeout, no retry, failure logged (per D-04).
    type WebhookClient struct {
        url    string
        format string // "json" or "base64"
        client *http.Client
    }

    // NewWebhookClient creates a webhook client. Empty url means no-op (per D-02).
    func NewWebhookClient(url, format string, timeout time.Duration) *WebhookClient {
        if format == "" {
            format = "json"
        }
        return &WebhookClient{
            url:    url,
            format: format,
            client: &http.Client{Timeout: timeout},
        }
    }

    // Fire sends the webhook POST. Non-blocking caller expected to call in goroutine.
    // Returns nil immediately if url is empty (per D-02).
    func (wc *WebhookClient) Fire(payload WebhookPayload) error {
        if wc.url == "" {
            return nil
        }

        var body []byte
        var contentType string

        switch wc.format {
        case "base64":
            jsonBytes, err := json.Marshal(payload)
            if err != nil {
                return fmt.Errorf("webhook marshal: %w", err)
            }
            encoded := base64.StdEncoding.EncodeToString(jsonBytes)
            body = []byte(encoded)
            contentType = "application/octet-stream"
        default: // "json"
            var err error
            body, err = json.Marshal(payload)
            if err != nil {
                return fmt.Errorf("webhook marshal: %w", err)
            }
            contentType = "application/json"
        }

        resp, err := wc.client.Post(wc.url, contentType, bytes.NewReader(body))
        if err != nil {
            return fmt.Errorf("webhook POST %s: %w", wc.url, err)
        }
        resp.Body.Close()
        return nil
    }
    ```

    ```go
    // HandlerConfig holds per-zone DSYNC configuration for the handler.
    type HandlerConfig struct {
        WebhookURL        string
        WebhookBodyFormat string
        AllowedSources    []string

    type Handler struct {
        limiter      *NotifyLimiter
        log          zerolog.Logger
        zoneACLs     map[string]*SourceACL      // zone FQDN -> ACL
        zoneWebhooks map[string]*WebhookClient  // zone FQDN -> webhook client
    }

    func NewHandler(limiter *NotifyLimiter, log zerolog.Logger, zones map[string]HandlerConfig) *Handler {
        h := &Handler{
            limiter:      limiter,
            log:          log.With().Str("component", "dsync").Logger(),
            zoneACLs:     make(map[string]*SourceACL),
            zoneWebhooks: make(map[string]*WebhookClient),
        }
        for zone, cfg := range zones {
            fqdn := dns.Fqdn(zone)
            h.zoneACLs[fqdn] = NewSourceACL(cfg.AllowedSources)
            if cfg.WebhookURL != "" {
                timeout := 5 * time.Second // D-04: fire-and-forget 5s timeout

        }
        return h
    }
    ```

    3. Modify `HandleInbound` to check source ACL BEFORE acknowledging (per D-06: REFUSED for rejected sources):

    ```go
    func (h *Handler) HandleInbound(w dns.ResponseWriter, r *dns.Msg, clientIP net.IP) {
        reply := new(dns.Msg)
        reply.SetReply(r)

        if len(r.Question) == 0 {
            reply.Rcode = dns.RcodeFormatError
            w.WriteMsg(reply)
            return
        }

        zone := dns.Fqdn(r.Question[0].Name)
        qtype := r.Question[0].Qtype

        // D-05/D-06: Check source ACL before anything else
        if acl, ok := h.zoneACLs[zone]; ok {
            if !acl.Check(clientIP) {
                reply.Rcode = dns.RcodeRefused
                w.WriteMsg(reply)
                return
            }
        }

        // D-14: Rate limit check — REFUSED on excess (consistent with D-06)
        if !h.limiter.Allow(clientIP) {
            reply.Rcode = dns.RcodeRefused
            w.WriteMsg(reply)
            return
        }

        // Acknowledge NOERROR per RFC 1996 section 3.10
        reply.Rcode = dns.RcodeSuccess
        w.WriteMsg(reply)

        // Only process CDS and CSYNC NOTIFY types per RFC 9859
        if qtype != dns.TypeCDS && qtype != dns.TypeCSYNC {
            return
        }

        // D-01: Fire webhook POST (fire-and-forget in goroutine)
        if wh, ok := h.zoneWebhooks[zone]; ok {
            go func() {
                payload := WebhookPayload{
                    Zone:      zone,
                    Qtype:     dns.TypeToString[qtype],
                    SourceIP:  clientIP.String(),
                    Timestamp: time.Now(),
                }
                if err := wh.Fire(payload); err != nil {
                    h.log.Error().Err(err).Str("zone", zone).Msg("webhook delivery failed")
                }
            }()
        }

        // Async delegation check
        go h.scheduleDelegationCheck(zone, qtype)
    }
    ```
