    ```go
    type Handler struct {
        limiter *NotifyLimiter
    }

    func NewHandler(limiter *NotifyLimiter) *Handler {
        return &Handler{limiter: limiter}
    }

    func (h *Handler) HandleInbound(w dns.ResponseWriter, r *dns.Msg, clientIP net.IP) {
        reply := new(dns.Msg)
        reply.SetReply(r)

        if len(r.Question) == 0 {
            reply.Rcode = dns.RcodeFormatError
            w.WriteMsg(reply)
            return
        }

        // Always send NOERROR acknowledgement per RFC 1996 §3.10
        // Even rate-limited requests get acknowledged (RFC 9859 §4.3 intent: 
        // do not reveal processing state to potential attackers)
        reply.Rcode = dns.RcodeSuccess
        w.WriteMsg(reply)

        // Rate limit check AFTER acknowledgement
        if !h.limiter.Allow(clientIP) {
            // Rate limited: acknowledged but not processed
            return
        }

        qtype := r.Question[0].Qtype
        zone  := r.Question[0].Name

        // Only process CDS and CSYNC NOTIFY types per RFC 9859
        if qtype != dns.TypeCDS && qtype != dns.TypeCSYNC {
            return
        }

        // Async delegation check (log only in this phase — no full maintenance engine yet)
        go h.scheduleDelegationCheck(zone, qtype)
    }

    func (h *Handler) scheduleDelegationCheck(zone string, qtype uint16) {
        // Phase 8 scope: log the notification received. Full delegation
        // maintenance engine (DS record updates) is deferred to a follow-on phase.
        qtypeName := dns.TypeToString[qtype]
        if qtypeName == "" {
            qtypeName = fmt.Sprintf("TYPE%d", qtype)
        }
        fmt.Printf("[dsync] NOTIFY(%s) received for zone %s — delegation check scheduled (stub)\n", qtypeName, zone)
    }
    ```

    ```go
    // DiscoverDSYNC queries _dsync.<zone> and parent labels for DSYNC records.
    // It uses the provided dns.Client and resolver address.
    // Returns all DSYNCRecord values found (may be empty — not an error).
    func DiscoverDSYNC(ctx context.Context, delegation string, client *dns.Client, resolver string) ([]DSYNCRecord, error) {
        labels := dns.SplitDomainName(dns.Fqdn(delegation))
        // Try _dsync.<delegation>, then _dsync.<parent>, etc.
        // Max depth: len(labels)-1 (stop before TLD-only)
        for i := 0; i < len(labels)-1; i++ {
            suffix := strings.Join(labels[i:], ".") + "."
            candidate := "_dsync." + suffix
            records, err := queryDSYNC(ctx, candidate, client, resolver)
            if err == nil && len(records) > 0 {
                return records, nil
            }
        }
        return nil, nil
    }

    // queryDSYNC sends a TYPE66 query and decodes any DSYNC records in the answer.
    func queryDSYNC(ctx context.Context, name string, client *dns.Client, resolver string) ([]DSYNCRecord, error) {
        m := new(dns.Msg)
        m.SetQuestion(name, TypeDSYNC)
        m.RecursionDesired = true
        resp, _, err := client.ExchangeContext(ctx, m, resolver)
        if err != nil {
            return nil, err
        }
        var results []DSYNCRecord
        for _, rr := range resp.Answer {
            if r3, ok := rr.(*dns.RFC3597); ok && r3.Hdr.Rrtype == TypeDSYNC {
                rec, err := ParseRFC3597(r3)
                if err != nil {
                    continue // skip malformed records
                }
                // RFC 9859: ignore records with Scheme=0 or Port=0
                if rec.Scheme == DSYNCSchemeNull || rec.Port == 0 {
                    continue
                }
                results = append(results, rec)
            }
        }
        return results, nil
    }
    ```

    ```go
    // DSYNCNotifier sends outbound NOTIFY(CDS/CSYNC) to discovered parent endpoints.
    // Events are enqueued on a channel; a worker goroutine drains it with propagation delay.
    type DSYNCNotifier struct {
        resolver         string        // DNS resolver address for _dsync discovery (e.g., "127.0.0.1:53")
        propagationDelay time.Duration // Wait before sending (default 60s per RFC 9859 recommendation)
        events           chan notifyEvent
    }

    type notifyEvent struct {
        zone  string
        qtype uint16
    }

    func NewDSYNCNotifier(resolver string, propagationDelay time.Duration) *DSYNCNotifier {
        n := &DSYNCNotifier{
            resolver:         resolver,
            propagationDelay: propagationDelay,
            events:           make(chan notifyEvent, 64),
        }
        go n.worker()
        return n
    }

    // Notify enqueues an outbound NOTIFY event for the given zone and qtype.
    // Non-blocking: if the queue is full, the event is dropped with a log line.
    func (n *DSYNCNotifier) Notify(zone string, qtype uint16) {
        select {
        case n.events <- notifyEvent{zone: zone, qtype: qtype}:
        default:
            fmt.Printf("[dsync] Notify queue full; dropping event for %s\n", zone)
        }
    }

    func (n *DSYNCNotifier) worker() {
        client := &dns.Client{Net: "udp", Timeout: 5 * time.Second}
        for ev := range n.events {
            // Apply propagation delay
            if n.propagationDelay > 0 {
                time.Sleep(n.propagationDelay)
            }
            ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
            records, err := DiscoverDSYNC(ctx, ev.zone, client, n.resolver)
            cancel()
            if err != nil || len(records) == 0 {
                continue
            }
            for _, rec := range records {
                // Only send for matching qtype
                if rec.RRtype != ev.qtype {
                    continue
                }
                if rec.Scheme != DSYNCSchemeNOTIFY {
                    continue
                }
                ctx2, cancel2 := context.WithTimeout(context.Background(), 5*time.Second)
                if err := sendNotify(ctx2, ev.zone, ev.qtype, rec.Target, rec.Port); err != nil {
                    fmt.Printf("[dsync] sendNotify %s: %v\n", rec.Target, err)
                }
                cancel2()
            }
        }
    }

    // sendNotify sends a single NOTIFY(qtype) to target:port.
    // Uses SetNotify (sets Opcode=4, AA=true) then overrides Qtype.
    func sendNotify(ctx context.Context, zoneName string, qtype uint16, target string, port uint16) error {
        m := new(dns.Msg)
        m.SetNotify(zoneName)
        m.Question[0].Qtype = qtype  // Override TypeSOA → TypeCDS or TypeCSYNC

        c := &dns.Client{Net: "udp", Timeout: 5 * time.Second}
        addr := net.JoinHostPort(strings.TrimSuffix(target, "."), strconv.Itoa(int(port)))
        resp, _, err := c.ExchangeContext(ctx, m, addr)
        if err != nil {
            return fmt.Errorf("notify %s: %w", addr, err)
        }
        if resp.Rcode != dns.RcodeSuccess {
            return fmt.Errorf("notify %s: rcode %s", addr, dns.RcodeToString[resp.Rcode])
        }
        return nil
    }
    ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| network → handler | Inbound NOTIFY from any IP; opcode + qtype are untrusted |
| handler → goroutine | scheduleDelegationCheck receives zone name from untrusted NOTIFY question |
| _dsync query response → discovery | Answer section may contain malformed DSYNC rdata |
| outbound sender → parent | Target FQDN from DSYNC record; port from DSYNC record |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-03 | DoS | HandleInbound / scheduleDelegationCheck | mitigate | goroutine spawned only AFTER rate limiter allows; limiter is applied before goroutine creation; rate-limited paths do NOT spawn goroutines |
| T-08-04 | Spoofing | DiscoverDSYNC / _dsync response | accept | DNSSEC validation of _dsync responses is RECOMMENDED by RFC 9859 but requires DNSSEC resolver support; deferred to follow-on phase; logged when records are accepted |
| T-08-05 | DoS | discovery loop (DiscoverDSYNC) | mitigate | Loop bounded by len(labels)-1; context.WithTimeout(10s) on each discovery call; no recursive referral following |
| T-08-06 | Tampering | sendNotify / outbound msg | accept | NOTIFY itself is unauthenticated per RFC 1996 base; parent validates CDS via DNSSEC; low risk |
| T-08-07 | DoS | DSYNCNotifier event queue | mitigate | Channel is buffered (64); Notify() drops events when full with a log line; worker goroutine applies propagation delay preventing burst |
